package Lab3

// Класи-заглушки даних користувача та токенів
type AuthToken struct {
        Token     string
        ExpiresAt time.Time
}

type User struct {
        Email        string
        PasswordHash string
}

// Головний сервіс автентифікації користувачів
type AuthService struct {
        usersDb      []User
        activeTokens map[string]AuthToken
}

func (s *AuthService) Authenticate(email string, pwd string) (*AuthToken, error) {
        if email == "" || pwd == "" {
                return nil, ErrEmptyCredentials
        }

        user := s.FindUserByEmail(email)
        if user == nil {
                return nil, ErrUserNotFound
        }

        incomingHash := s.HashPassword(pwd)
        if user.PasswordHash != incomingHash {
                return nil, ErrInvalidPassword
        }

        token := &AuthToken{
                Token:     GenerateGuid(),
                ExpiresAt: time.Now().Add(2 * time.Hour),
        }
        s.activeTokens[token.Token] = *token
        return token, nil
}

func (s *AuthService) Register(email string, pwd string) (*User, error) {
        if email == "" || len(pwd) < 6 {
                return nil, ErrInvalidRegistrationData
        }

        if s.UserExists(email) {
                return nil, ErrUserAlreadyExists
        }

        newUser := &User{
                Email:        email,
                PasswordHash: s.HashPassword(pwd),
        }
        s.usersDb = append(s.usersDb, *newUser)
        return newUser, nil
}

func (s *AuthService) ValidateToken(token *AuthToken) bool {
        if token == nil || token.Token == "" {
                return false
        }

        foundToken, exists := s.activeTokens[token.Token]
        if exists {
                if foundToken.ExpiresAt.After(time.Now()) {
                        return true
                } else {
                        delete(s.activeTokens, token.Token)
                }
        }
        return false
}

func (s *AuthService) HashPassword(pwd string) string {
        if pwd == "" {
                return ""
        }
        return ComputeSha256(pwd)
}
